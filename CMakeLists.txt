cmake_minimum_required(VERSION 3.20)
project(pex_window_tests LANGUAGES C)
include(CTest)
option(ENABLE_SANITIZERS "Run tests with address and undefined-behavior sanitizers" OFF)
if(BUILD_TESTING)
  add_executable(test_plan tests/test_plan.c)
  target_compile_features(test_plan PRIVATE c_std_11)
  target_compile_options(test_plan PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow -UNDEBUG)
  if(ENABLE_SANITIZERS)
    target_compile_options(test_plan PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(test_plan PRIVATE -fsanitize=address,undefined)
  endif()
  foreach(case five_gpus nested_branches boundaries conflicts capacity many_switches shared_aperture fragmentation)
    add_test(NAME plan.${case} COMMAND test_plan ${case})
  endforeach()
endif()
