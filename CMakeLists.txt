cmake_minimum_required(VERSION 3.20)
project(pex_window_tests LANGUAGES C)
include(CTest)
option(ENABLE_SANITIZERS "Run tests with address and undefined-behavior sanitizers" OFF)
if(BUILD_TESTING)
  foreach(target test_plan test_transaction)
    add_executable(${target} tests/${target}.c)
    target_compile_features(${target} PRIVATE c_std_11)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow -UNDEBUG)
    if(ENABLE_SANITIZERS)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endforeach()
  foreach(case five_gpus nested_branches boundaries conflicts capacity many_switches shared_aperture fragmentation)
    add_test(NAME plan.${case} COMMAND test_plan ${case})
  endforeach()
  foreach(case verified_command dma_ordering apply_failures rollback_failure rollback_destination rollback_fallback)
    add_test(NAME transaction.${case} COMMAND test_transaction ${case})
  endforeach()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME installer.recovery COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_install.py)
endif()
find_package(Doxygen QUIET)
if(Doxygen_FOUND)
  add_custom_target(docs
    COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_SOURCE_DIR}/build
    COMMAND ${DOXYGEN_EXECUTABLE} ${CMAKE_SOURCE_DIR}/Doxyfile
    WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
    COMMENT "Build API documentation with warnings treated as errors")
endif()
